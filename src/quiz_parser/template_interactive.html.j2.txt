<script>
const quizRawData = {{ quiz | tojson }};
const imgMap = {{ img_map | tojson }};
const audioMap = {{ audio_map | tojson }};
const shuffleFlag = {{ shuffle | lower }};

let correctSet = new Set();
let wrongSet = new Set();
let questionList = [];

// 解析原始groups扁平化题目
function flattenQuestions(rawQuiz){
    const qs = [];
    for(const g of rawQuiz.groups){
        for(const q of g.questions){
            qs.push({
                groupTitle: g.gt,
                groupAudioKey: g._audio_logic_path ?? null,
                groupImagePaths: g._image_logic_paths ?? [],
                question: q
            })
        }
    }
    return qs;
}

function shuffleArray(arr){
    const copy = [...arr];
    for(let i=copy.length‑1; i>0;i--){
        const j = Math.floor(Math.random()*(i+1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
    }
    return copy;
}

function updateScore(){
    const total = questionList.length;
    const ok = correctSet.size;
    const err = wrongSet.size;
    const pct = total>0 ? Math.round((ok/total)*100):0;
    document.getElementById("score").textContent = pct;
    document.getElementById("cnt‑ok").textContent = ok;
    document.getElementById("cnt‑err").textContent = err;
}

function renderOneQuestion(qWrapper, idx){
    const root = document.getElementById("quiz‑root");
    const q = qWrapper.question;
    const div = document.createElement("div");
    div.className = "question‑container";
    div.dataset.qid = String(idx);

    let audioHtml = "";
    // group顶层音频
    if(qWrapper.groupAudioKey && audioMap[qWrapper.groupAudioKey]){
        audioHtml += `<div class="audio‑wrap"><audio controls src="${audioMap[qWrapper.groupAudioKey]}"></div>`;
    }
    // 题目自身a音频
    if(q._audio_logic_path && audioMap[q._audio_logic_path]){
        audioHtml += `<div class="audio‑wrap"><audio controls src="${audioMap[q._audio_logic_path]}"></div>`;
    }

    let imgHtml = "";
    // group级图片
    for(const sp of qWrapper.groupImagePaths){
        if(imgMap[sp]){
            imgHtml += `<img class="q‑img" src="${imgMap[sp]}">`;
        }
    }
    // 题目图片
    const qImgPaths = q._image_logic_paths ?? [];
    for(const sp of qImgPaths){
        if(imgMap[sp]){
            imgHtml += `<img class="q‑img" src="${imgMap[sp]}">`;
        }
    }

    let optHtml = "";
    if(q.o){
        const alpha = ["A","B","C","D","E","F"];
        q.o.forEach((opt, oIdx)=>{
            optHtml += `
            <div class="opt‑row" data‑opt‑key="${opt.k}" data‑qid="${idx}">
                <input type="radio" name="q_${idx}" id="opt_${idx}_${oIdx}" value="${opt.k}">
                <label for="opt_${idx}_${oIdx}">${opt.t}</label>
            </div>`
        })
    }

    div.innerHTML = `
        <div class="q‑header‑row">
            <div class="q‑label">题目 ${idx+1}</div>
            <button class="btn‑explain" data‑qidx="${idx}">查看解析</button>
        </div>
        <div class="q‑stem">${q.stem ?? ""}</div>
        ${audioHtml}
        ${imgHtml}
        <div class="options‑wrap">${optHtml}</div>
    `;
    root.appendChild(div);
}

function bindEvents(){
    // 解析弹窗
    document.querySelectorAll(".btn‑explain").forEach(btn=>{
        btn.onclick = (ev)=>{
            const qIdx = Number(ev.target.dataset.qidx);
            const modal = document.getElementById("modal");
            const modalText = document.getElementById("modal‑text");
            const qObj = questionList[qIdx].question;
            modalText.innerText = qObj.e ?? "暂无解析";
            modal.showModal();
        }
    })
    // 单选点击判分
    document.querySelectorAll(".opt‑row input[type='radio']").forEach(radio=>{
        radio.onchange = (ev)=>{
            const inp = ev.target;
            const qid = inp.closest(".opt‑row").dataset.qid;
            const userSel = inp.value;
            const targetQ = questionList[Number(qid)].question;
            const correctKey = targetQ.ka;

            // 清除该题旧标记
            document.querySelectorAll(`.opt‑row[data‑qid="${qid}"]`).forEach(r=>{
                r.classList.remove("correct‑mark","wrong‑mark");
            })
            if(userSel === correctKey){
                correctSet.add(qid);
                wrongSet.delete(qid);
                inp.closest(".opt‑row").classList.add("correct‑mark");
            }else{
                wrongSet.add(qid);
                correctSet.delete(qid);
                inp.closest(".opt‑row").classList.add("wrong‑mark");
            }
            updateScore();
        }
    })
}

function main(){
    let raw = flattenQuestions(quizRawData);
    if(shuffleFlag){
        raw = shuffleArray(raw);
    }
    questionList = raw;
    const root = document.getElementById("quiz‑root");
    root.innerHTML = "";
    questionList.forEach((item,idx)=>{
        renderOneQuestion(item, idx);
    })
    bindEvents();
    updateScore();
}
